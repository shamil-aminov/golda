/** The one rule: a usual day is the median of the days with spending, an outlier is ten of them or more. */
private fun outliers(values: List<Long>): Set<Long> {
    val days = values.filter { it > 0 }.sorted()
    if (days.isEmpty()) return emptySet()
    val usual = if (days.size % 2 == 1) days[days.size / 2] else (days[days.size / 2 - 1] + days[days.size / 2]) / 2
    return days.filter { it >= usual * 10 }.toSet()
}

private class Day(val date: LocalDate, val value: Long, val parts: List<DayPart>)

@Composable
private fun daysOf(data: AppData, v: V): List<Day> {
    val tones = LocalTones.current
    val period = Analytics.period(v.kind, data.today)
    val only = data.operations.filter { it.op.type == OpType.EXPENSE && (v.barsFilter == null || Buckets.of(it.op, data.categoryById) == v.barsFilter) }
    val report = Analytics.report(only, data.accountById, data.categoryById, period, data.today, data.zone, data.rates)
    val stacks = only.groupBy { Ledger.localDate(it.op.timestamp, data.zone) }.mapValues { (_, ops) ->
        ops.groupBy { it.op.categoryId?.let(data.categoryById::get) }.map { (cat, list) ->
            DayPart(cat?.key?.let { categoryColor(it, tones) } ?: leitmotifColor(null, tones), cat?.label() ?: "Без категории", -list.sumOf { o -> o.postings.sumOf { p -> p.rubMinor } })
        }.filter { it.rub > 0 }.sortedByDescending { it.rub }
    }
    return report.days.map { (d, rub) -> Day(d, rub, stacks[d].orEmpty()) }
}

/** Weeks counted back from today, so the last bar is a whole week and the short one is the oldest. */
private fun weeks(days: List<Day>): List<Day> = days.reversed().chunked(7).reversed().map { week ->
    val list = week.reversed()
    Day(list.first().date, list.sumOf { it.value }, list.flatMap { it.parts }.groupBy { it.name }.map { (_, p) -> p.first().copy(rub = p.sumOf { it.rub }) }.sortedByDescending { it.rub })
}

private fun DrawScope.bar(v: V, left: Float, width: Float, h: Float, parts: List<DayPart>, floor: Color, neutral: Color, card: Color, cut: Boolean) {
    val r = when (v.corner) {
        Corner.DOME -> minOf(width / 2, 12.dp.toPx())
        Corner.TOP4, Corner.BOTH4 -> 4.dp.toPx()
        Corner.TOP8 -> 8.dp.toPx()
        Corner.SQUARE -> 0f
        Corner.PROP -> (minOf(width, h) * 0.12f).coerceIn(4.dp.toPx(), 16.dp.toPx())
    }.coerceAtMost(width / 2)
    val top = size.height - h
    val low = if (v.corner == Corner.BOTH4) CornerRadius(r) else CornerRadius.Zero
    val path = Path().apply { addRoundRect(RoundRect(left, top, left + width, size.height, CornerRadius(r), CornerRadius(r), low, low)) }
    val sum = parts.sumOf { it.rub }
    when {
        sum == 0L -> drawPath(path, floor)
        v.sep == Sep.BIG -> drawPath(path, parts.first().color)
        v.sep == Sep.NEUTRAL -> drawPath(path, neutral)
        v.sep == Sep.TILES -> {
            var y = size.height
            val gap = 2.dp.toPx()
            val room = h - gap * (parts.size - 1)
            parts.forEach { p ->
                val part = (room * p.rub / sum).coerceAtLeast(3.dp.toPx())
                drawRoundRect(p.color, Offset(left, y - part), Size(width, part), CornerRadius(minOf(r, part / 2)))
                y -= part + gap
            }
        }
        else -> clipPath(path) {
            var y = size.height
            val gap = if (v.sep == Sep.HAIR) 1.dp.toPx() else 0f
            parts.forEach { p ->
                val part = h * p.rub / sum
                drawRect(p.color, Offset(left, y - part), Size(width, (part - gap).coerceAtLeast(0f)))
                y -= part
            }
        }
    }
    if (cut && v.cut == Cut.MARK) {
        // Two strokes of the card across the bar: the break.
        val y = top + 16.dp.toPx()
        listOf(0f, 6.dp.toPx()).forEach { d ->
            drawLine(card, Offset(left - 2.dp.toPx(), y + d + 4.dp.toPx()), Offset(left + width + 2.dp.toPx(), y + d - 4.dp.toPx()), strokeWidth = 2.dp.toPx())
        }
    }
    if (cut && v.cut == Cut.FADE) {
        drawRect(Brush.verticalGradient(listOf(card, card.copy(alpha = 0f)), top, top + 36.dp.toPx()), Offset(left - 1f, top - 1f), Size(width + 2f, 37.dp.toPx()))
    }
}

@Composable
private fun Bars(v: V, days: List<Day>, base: sh.aminov.golda.domain.Base) {
    val floor = MaterialTheme.colorScheme.outlineVariant
    val neutral = MaterialTheme.colorScheme.onSurface.copy(alpha = 0.28f)
    val card = tileColor()
    val density = LocalDensity.current
    val high = 96.dp
    val out = if (v.cut == Cut.OFF) emptySet() else outliers(days.map { it.value })
    val rest = days.filter { it.value !in out }.maxOf { it.value }.coerceAtLeast(1)
    val top = days.maxOf { it.value }.coerceAtLeast(1)
    // The ceiling a little over the tallest usual day, so a cut bar still stands out.
    val ceiling = if (out.isEmpty()) top else (rest * 1.25f).toLong()
    BoxWithConstraints(Modifier.fillMaxWidth()) {
        val full = maxWidth
        val w = with(density) { full.toPx() }
        val n = days.size
        val gapPx = v.gap?.let { with(density) { it.toPx() } }
        val step = when {
            v.shape == Shape.SCROLL -> w / 10.5f
            gapPx != null -> (w + gapPx) / n
            else -> w / n
        }
        val barW = if (gapPx != null) step - gapPx else step * 0.8f
        fun left(i: Int) = if (v.shape == Shape.SCROLL) w - step * (n - i) + (step - barW) else step * i + if (gapPx != null) 0f else (step - barW) / 2
        Column {
            Box(Modifier.fillMaxWidth().height(48.dp)) {
                days.forEachIndexed { i, d ->
                    if (d.value in out) {
                        val wide = 96.dp
                        val x = with(density) { (left(i) + barW / 2).toDp() } - wide / 2
                        Text(
                            base.whole(d.value), Modifier.offset(x = x.coerceIn(0.dp, full - wide), y = 26.dp).width(wide),
                            style = TextRole.caption, color = MaterialTheme.colorScheme.onSurface, textAlign = TextAlign.Center, maxLines = 1,
                        )
                    }
                }
            }
            Canvas(Modifier.fillMaxWidth().height(high)) {
                clipPath(Path().apply { addRect(androidx.compose.ui.geometry.Rect(0f, -40f, size.width, size.height + 40f)) }) {
                    days.forEachIndexed { i, d ->
                        val share = when {
                            v.root -> sqrt(d.value.toFloat() / top)
                            else -> minOf(d.value, ceiling).toFloat() / ceiling
                        }
                        val h = if (d.value > 0) (size.height * share).coerceAtLeast(4.dp.toPx()) else 3.dp.toPx()
                        val x = left(i)
                        if (x + barW > 0) bar(v, x, barW, if (d.value in out) size.height else h, d.parts, floor, neutral, card, d.value in out)
                    }
                }
            }
            VSpace(Gap.m)
            if (n <= 7 && v.shape == Shape.DAYS) {
                Box(Modifier.fillMaxWidth().height(16.dp)) {
                    days.forEachIndexed { i, d ->
                        Text(
                            d.date.dayOfWeek.getDisplayName(TextStyle.SHORT_STANDALONE, I18n.locale).take(2).replaceFirstChar { it.titlecase(I18n.locale) },
                            Modifier.offset(x = with(density) { (left(i) + barW / 2).toDp() } - 28.dp).width(56.dp),
                            style = MaterialTheme.typography.labelSmall, color = MaterialTheme.colorScheme.onSurfaceVariant, textAlign = TextAlign.Center, maxLines = 1,
                        )
                    }
                }
            } else if (v.shape == Shape.WEEKS) {
                Box(Modifier.fillMaxWidth().height(16.dp)) {
                    days.forEachIndexed { i, d ->
                        Text(
                            dayMonth(d.date), Modifier.offset(x = with(density) { (left(i) + barW / 2).toDp() } - 36.dp).width(72.dp),
                            style = MaterialTheme.typography.labelSmall, color = MaterialTheme.colorScheme.onSurfaceVariant, textAlign = TextAlign.Center, maxLines = 1,
                        )
                    }
                }
            } else {
                Row(Modifier.fillMaxWidth()) {
                    Text(dayMonth(if (v.shape == Shape.SCROLL) days[n - 10].date else days.first().date), Modifier.weight(1f), style = TextRole.caption)
                    Text(dayMonth(days.last().date), style = TextRole.caption)
                }
            }
        }
    }
}


/** М4: the period as a calendar, a row a week, a day's depth by how much of the period's biggest it is, in its biggest category's colour. */
@Composable
private fun Calendar(days: List<Day>) {
    val soft = LocalTones.current.soft
    val top = days.maxOf { it.value }.coerceAtLeast(1)
    val lead = days.first().date.dayOfWeek.value - 1
    val cells = List(lead) { null } + days
    Column(verticalArrangement = Arrangement.spacedBy(4.dp)) {
        VSpace(Gap.m)
        cells.chunked(7).forEach { week ->
            Row(horizontalArrangement = Arrangement.spacedBy(4.dp)) {
                (0 until 7).forEach { i ->
                    val d = week.getOrNull(i)
                    val colour = when {
                        d == null -> Color.Transparent
                        d.value == 0L -> soft
                        else -> lerp(soft, d.parts.first().color, (0.35f + 0.65f * sqrt(d.value.toFloat() / top)).coerceAtMost(1f))
                    }
                    Box(Modifier.weight(1f).height(22.dp).background(colour, RoundedCornerShape(6.dp)))
                }
            }
        }
    }
}

@RunWith(ParameterizedRobolectricTestRunner::class)
@GraphicsMode(GraphicsMode.Mode.NATIVE)
@Config(qualifiers = "w411dp-h270dp-xxhdpi")
class BarsVariantsPreviewTest(private val name: String) {
    companion object {
        @JvmStatic @ParameterizedRobolectricTestRunner.Parameters(name = "{0}")
        fun names() = cases.keys.map { arrayOf<Any>(it) }
    }

    @get:Rule val compose = createComposeRule()

    @Before
    fun fixTheWorld() {
        TimeZone.setDefault(TimeZone.getTimeZone("UTC"))
        I18n.russian = true
    }

    @Test fun shot() {
        val data = ScreenshotTest().demo()
        val v = cases.getValue(name)
        compose.setContent {
            GoldaTheme("golda", "LIGHT", "ROOMY") {
                Box(Modifier.fillMaxSize().background(LocalTones.current.page).padding(Gap.s)) {
                    Surface(Modifier.fillMaxWidth(), shape = RoundedCornerShape(CardCorner), color = tileColor()) {
                        Column(Modifier.padding(start = 16.dp, end = 16.dp, top = 26.dp, bottom = 26.dp)) {
                            val title = v.filter?.let(::bucketLabel) ?: "Все траты"
                            Text(title, Modifier.letters(TextRole.groupTitle.fontSize), style = TextRole.groupTitle)
                            val days = daysOf(data, v)
                            if (name == "m0") java.io.File("build/bars-values.txt").writeText(days.filter { it.value > 0 }.joinToString("\n") { "${it.date} ${it.value / 100}" } + "\nout " + outliers(days.map { it.value }))
                            when (v.shape) {
                                Shape.CALENDAR -> Calendar(days)
                                Shape.WEEKS -> Bars(v, weeks(days), data.base)
                                else -> Bars(v, days, data.base)
                            }
                        }
                    }
                }
            }
        }
        compose.waitForIdle()
        compose.onRoot().captureRoboImage("../docs/tasks/analytics-contours/bars/$name.png")
    }
}

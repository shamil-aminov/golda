@Composable
private fun IslandForFrames() = GoldaToolbar(
    tabs = listOf(
        ToolbarTab(R.drawable.ic_home, R.drawable.ic_home_fill, "Главная"),
        ToolbarTab(R.drawable.ic_wallet, R.drawable.ic_wallet_fill, "Счета"),
        ToolbarTab(R.drawable.ic_chart, R.drawable.ic_chart_fill, "Аналитика"),
    ),
    selected = 2, position = { 2f }, onSelect = {}, voice = VoiceControl(VoiceState.Idle, {}), onAdd = {},
)

private val frames = mapOf(
    "day-b" to "", "day-s" to "sub", "day-r" to "right", "week-b" to "", "week-s" to "sub", "kind-b" to "", "kind-s" to "sub",
    "kd-b" to "", "kd-l" to "line", "kd-s" to "sub", "z-b" to "", "z-n" to "nozero", "z-1" to "one",
)

/** TEMPORARY, not committed. */
@RunWith(ParameterizedRobolectricTestRunner::class)
@GraphicsMode(GraphicsMode.Mode.NATIVE)
@Config(qualifiers = "w411dp-h891dp-xxhdpi")
class InsightsFramesPreviewTest(private val name: String) {
    companion object {
        @JvmStatic @ParameterizedRobolectricTestRunner.Parameters(name = "{0}")
        fun cases() = frames.keys.map { arrayOf<Any>(it) }
    }

    @get:Rule val compose = createComposeRule()

    @Before
    fun fixTheWorld() {
        TimeZone.setDefault(TimeZone.getTimeZone("UTC"))
        I18n.russian = true
        System.setProperty("golda.f", frames.getValue(name))
    }

    @After fun back() { System.clearProperty("golda.f") }

    private fun week() {
        compose.onNodeWithContentDescription("Период").performClick()
        compose.onNodeWithText("Неделя").performClick()
    }
    private fun bar(x: Float) = compose.onNodeWithTag(Tags.INSIGHTS_BARS).performTouchInput { click(Offset(width * x, height / 2f)) }

    @Test fun shot() {
        val data = ScreenshotTest().demo()
        compose.setContent {
            GoldaTheme("golda", "LIGHT", "ROOMY") {
                Scaffold(floatingActionButton = { IslandForFrames() }, floatingActionButtonPosition = FabPosition.Center) { padding ->
                    AnalyticsScreen(data, padding) {}
                }
            }
        }
        when (name.substringBefore('-')) {
            "day" -> { week(); bar(0.97f) }
            "week" -> bar(0.69f)
            "kind" -> compose.onNodeWithText("Регулярные").performClick()
            "kd" -> { week(); compose.onNodeWithText("Ежедневные").performClick(); bar(0.97f) }
            "z" -> if (name != "z-1") week()
        }
        compose.waitForIdle()
        compose.onRoot().captureRoboImage("../docs/tasks/analytics-contours/title/$name.png")
    }
}
